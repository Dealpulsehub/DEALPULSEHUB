"""
NEURO AD AUDITOR
Audita TODOS los aspectos psychology de ads/mockups

class NeuroAdAuditor:
    def audit_all_designs(self, design_dir="./design_assets"):
        """
        Audita para TODOS los designs:
        - Color psychology
        - Contrast (WCAG 4.5:1)
        - Visual hierarchy
        - Urgency signals
        - Psychology triggers
        - Accessibility
        """
        
        audit_results = []
        
        for design in designs:
            audit = {
                "file": design,
                "color_psychology": self.check_colors(design),
                "contrast": self.check_contrast(design),
                "hierarchy": self.check_hierarchy(design),
                "urgency": self.check_urgency(design),
                "psychology_triggers": self.check_triggers(design),
                "accessibility": self.check_wcag(design),
                "overall_verdict": self.overall_score(design),
                "recommendations": self.get_recommendations(design)
            }
            audit_results.append(audit)
        
        # Generate report
        self.generate_audit_report(audit_results)
        return audit_results
